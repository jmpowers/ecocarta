group_by_cell <- function(dat.points, grid, min_obs=0) {
  #assign points to cells
  dat.grid.cells <- st_intersects(grid, dat.points) 
  grid$n_obs <- lengths(dat.grid.cells) #count points in each cell
  
  #drop cells with few observations
  cells.full <- which(grid$n_obs > min_obs)
  grid.full <- grid[cells.full,]
  
  #tally number of each species in each cell
  dat.grid <- t(sapply(dat.grid.cells, function(x) table(dat.points$species[x])))
  rownames(dat.grid) <- grid$grid_id
  dat.grid.full <- dat.grid[cells.full,]
  
  list(data = dat.grid.full, grid = grid.full)
}

make_grid <- function(dat.points, box, 
                      method = c("hex_grid", "hulls_balanced", "voronoi_balanced", "voronoi_random"),
                      random_seed = 1, n_seeds = 200, min_obs = 8,
                      hex_dim = round(sqrt(n_seeds)), square = F, ratio = 2/(1 + sqrt(5))) {
  set.seed(random_seed)
  method <- match.arg(method)
  if(method == "hex_grid") {
    grid <- st_make_grid(dat.points, n = c(hex_dim, hex_dim), square = square) #square=F means hexagons
  } 
  if(method %in% c("hulls_balanced","voronoi_balanced")) {
    clust.points <- dat.points %>% 
      slice_head(n = floor(nrow(.)/n_seeds)*n_seeds) %>% #drop points so they can be evenly divided
      mutate(grid_id = balanced_clustering(st_coordinates(.), K = n_seeds))#algorithm makes equal-sized clusters
    grid <- clust.points %>% 
      summarise(geometry = st_concave_hull(st_union(geometry), ratio = if_else(method=="voronoi_balanced", 1, ratio)), .by=grid_id) %>% 
      terra::vect() %>% terra::erase() %>% st_as_sf() #prevent polygons from intersecting others
  }
  if(method == "voronoi_balanced") {
    grid <- grid %>% st_centroid() %>% st_combine() %>% 
      st_voronoi(point_order=T) %>% st_cast() %>% st_sf() #voronoi using centroids of balanced clustering polygons
  } 
  if(method == "voronoi_random") {
    grid <- sample_n(dat.points, n_seeds) %>% st_combine() %>% 
      st_voronoi() %>% st_cast() %>% st_sf() #voronoi using randomly selected points
  }
  grid <- grid %>% st_intersection(box) %>% st_make_valid() %>% st_sf() %>% mutate(grid_id = row_number())
  group_by_cell(dat.points, grid, min_obs = min_obs)
}

Divide into areas of similar population

The number of observations in the hex grid is too uneven. Instead, try to divide the area into polygons of more even populations. First, sample some random seed observations - dense observations areas should have more sampled points. Then, define the region that is closest to each point, making a Voronoi diagram. The relative variation in observation number per cell is now half that of the hex grid. The second Voronoi partitioning is based on the centers of polygons created in the next section.

vor.rand <- make_grid(dat.points, box, method = "voronoi_random")
plot_counts(vor.rand)

vor.bal <- make_grid(dat.points, box, method = "voronoi_balanced")
plot_counts(vor.bal)

Use an anticlustering algorithm to divide the observations into equal-sized clusters. Then draw a convex polygon around each cluster. Areas with no observations are clear now.

Cluster similar communities

Look at the observed community composition in each cell, defined as the relative abundance of each taxon. Assign each cell to a cluster of similar communities using the k-means algorithm. Set the number of clusters and assign each a color.

Species that vary across space

To visualize the similarity between communities in each Voronoi cell, run an NMDS ordination on relative abundances. Each point represents a cell, with cells of similar communities clustering together. The colors for each cluster are the same as above. Common species are labeled near the cells where they are most observed and colored by their taxonomic order. The communities are broadly organized by phylum.

plot_nmds <- function(data, grid, min_to_show) {
  nmds <- metaMDS(data, trace=F)
  
  par(bg="black", mar=c(0,0,0,0))
  plot(nmds, type="n", axes=F, ann=F)
  points(nmds, display="sites", col=pal[grid$cluster], pch=19, cex=1.5)
  text(nmds, display="species", cex=0.7, font=4,
       col=if_else(specs$n_total > min_to_show, phylum_pal[specs$phylum], alpha("black",0)))
  legend("topleft", levels(specs$phylum), fill=phylum_pal, bty="n", text.col="white")
  par(bg="white")
}
plot_nmds(dat.tri, tri.full, 300)

Arrange clusters along gradient

Try to order the clusters by similarity in a linear gradient. This could potentially capture an urban to forested gradient, or riparian to upland, or what people are interested in for different areas. To do this, run a 1 dimensional NMDS ordination of the centers of each cluster, which orders clusters along the number line. Then map the ordered cluster number to a color scale.

tri.km.nmds <- metaMDS(tri.km$centers, k=1, trace=F)
tri.full$cluster_ordered <- factor(as.integer(factor(tri.full$cluster, levels=order(t(tri.km.nmds$points)))))
ggplot() + geom_sf(data = tri.full, aes(fill = cluster_ordered), linewidth=0.5, color="grey80") + 
  theme_void() + scale_fill_viridis_d("Cluster", option="mako", guide="none")
